> For Claude: REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

## Goal

Integrate the React dashboard (`dashboard/`) with the Rust Axum API so the dashboard displays **real engine data** instead of mocks:

1. **Backend**: expose graph/lifecycle/AIS/SSE endpoints on the existing API, add CORS, and publish an OpenAPI spec with Swagger UI.
2. **Frontend**: generate a typed TS client from the OpenAPI spec, wire React Query hooks + SSE live events into the existing components, and keep a **"Sim Mode"** fallback (existing `mockData`) whenever the backend is unreachable.
3. **Quality gates**: `cargo fmt/clippy/test/doc`, `tsc --noEmit`, `vitest`, `npm run build`, and a CI step that fails if `openapi.json` is out of sync with the code.

    /// List all unique undirected edges (`from < to`, deduplicated).
    ///
    /// Deterministic order (sorted by `from`, then `to`).
    pub fn edges(&self) -> Vec<Edge> {
        let mut seen: BTreeSet<(u64, u64)> = BTreeSet::new();
        let mut out = Vec::new();
        for (&from, neighbors) in &self.adjacency {
            for &(to, weight) in neighbors {
                if from < to && seen.insert((from, to)) {
                    out.push(Edge { from, to, weight });
                }
            }
        }
        out
    }

    /// Remove a node and every edge that references it.
    ///
    /// Idempotent: unknown ids are ignored.
    pub fn remove_node(&mut self, id: u64) {
        self.adjacency.remove(&id);
        for neighbors in self.adjacency.values_mut() {
            neighbors.retain(|(n, _)| *n != id);
        }
    }
```

    pub fn list_nodes(&self) -> Vec<EcmaNode> {
        self.node_registry.read().values().cloned().collect()
    }

    /// List all unique undirected edges of the semantic graph.
    pub fn graph_edges(&self) -> Vec<kce_graph::graph::Edge> {
        self.graph.read().edges()
    }

    /// Snapshot of pheromone levels keyed by `(from, to)` exactly as deposited.
    pub fn pheromone_levels(&self) -> HashMap<(u64, u64), f64> {
        self.pheromone_map
            .read()
            .all_entries()
            .iter()
            .map(|(k, v)| (*k, v.level))
            .collect()
    }

    /// Insert a node, or merge into an existing one (increments `usage_count`,
    /// takes latest `state`/`entropy`/`maturity`).
    pub fn upsert_node(&self, node: EcmaNode) {
        let mut reg = self.node_registry.write();
        match reg.entry(node.id) {
            std::collections::btree_map::Entry::Occupied(mut occ) => {
                let existing = occ.get_mut();
                existing.usage_count += 1;
                existing.state = node.state;
                existing.entropy = node.entropy;
                existing.maturity = node.maturity;
                existing.connections = node.connections;
            }
            std::collections::btree_map::Entry::Vacant(vac) => {
                vac.insert(node);
            }
        }
    }

    /// Set a node's lifecycle state. Returns `None` if the node is unknown.
    pub fn set_node_state(&self, id: u64, state: NodeState) -> Option<EcmaNode> {
        self.node_registry.write().get_mut(&id).map(|n| {
            n.state = state;
            *n
        })
    }

    /// Connect two registered nodes (bidirectional edge). Returns `false`
    /// when either node is not registered.
    pub fn connect(&self, from: u64, to: u64, weight: f64) -> bool {
        {
            let reg = self.node_registry.read();
            if !reg.contains_key(&from) || !reg.contains_key(&to) {
                return false;
            }
        }
        let mut graph = self.graph.write();
        graph.add_edge(from, to, weight);
        let from_edges = graph.neighbors(from).len();
        let to_edges = graph.neighbors(to).len();
        drop(graph);
        let mut reg = self.node_registry.write();
        if let Some(n) = reg.get_mut(&from) {
            n.connections = from_edges;
        }
        if let Some(n) = reg.get_mut(&to) {
            n.connections = to_edges;
        }
        true
    }

    /// Remove a node from the registry and the graph. Returns `false` when
    /// the node was not registered (graph cleanup still runs).
    pub fn remove_node(&self, id: u64) -> bool {
        let removed = self.node_registry.write().remove(&id).is_some();
        self.graph.write().remove_node(id);
        removed
    }
```

//! Engine event bus (SSE payloads).

use serde::Serialize;

/// Broadcast channel capacity for engine events.
pub const EVENT_CHANNEL_CAPACITY: usize = 64;

/// Events emitted by the engine and streamed over `GET /events`.
#[derive(Debug, Clone, Serialize)]
#[serde(tag = "type", rename_all = "snake_case")]
pub enum EngineEvent {
    /// A query completed successfully.
    QueryCompleted {
        /// End-to-end pipeline latency in milliseconds.
        latency_ms: f64,
        /// Executed action name (e.g. `classify`).
        action: String,
    },
    /// A node was created via `POST /nodes`.
    NodeCreated {
        /// New node id.
        id: u64,
    },
    /// A node was removed via `DELETE /nodes/:id`.
    NodeRemoved {
        /// Removed node id.
        id: u64,
    },
    /// Pheromone was deposited via `POST /pheromone/deposit`.
    PheromoneDeposited {
        /// Source node.
        from: u64,
        /// Target node.
        to: u64,
        /// Resulting pheromone level.
        level: f64,
    },
    /// AIS anomaly detected by `POST /ais/probe`.
    AisAnomaly {
        /// Anomaly score.
        score: f64,
        /// Whether the vector would be blocked.
        blocked: bool,
    },
}

/// Handle type for the event broadcast channel.
pub type EngineEventTx = tokio::sync::broadcast::Sender<EngineEvent>;

/// Full semantic graph: nodes, edges, pheromone levels.
async fn graph_handler(State(state): State<Arc<AppState>>) -> (StatusCode, Json<GraphResponse>) {
    let orchestrator = &state.orchestrator;
    let nodes = orchestrator
        .list_nodes()
        .into_iter()
        .map(|n| GraphNodeDto {
            id: n.id,
            state: n.state,
            usage_count: n.usage_count,
            entropy: n.entropy,
            connections: n.connections,
            maturity: n.maturity,
        })
        .collect();
    let edges = orchestrator
        .graph_edges()
        .into_iter()
        .map(|e| GraphEdgeDto {
            from: e.from,
            to: e.to,
            weight: e.weight,
        })
        .collect();
    // Keys are stored exactly as deposited; expose both orientations so the
    // client can look up (from,to) regardless of deposit direction.
    let mut pheromone = std::collections::HashMap::new();
    for ((from, to), level) in orchestrator.pheromone_levels() {
        pheromone.insert(format!("{from}_{to}"), level);
        pheromone.insert(format!("{to}_{from}"), level);
    }
    (StatusCode::OK, Json(GraphResponse { nodes, edges, pheromone }))
}

/// Create a node (optionally connected to an existing one).
async fn create_node_handler(
    State(state): State<Arc<AppState>>,
    Json(req): Json<CreateNodeRequest>,
) -> Result<(StatusCode, Json<CreateNodeResponse>), (StatusCode, Json<crate::routes::ErrorBody>)> {
    let weight = req.weight.unwrap_or(0.5);
    if !weight.is_finite() || !(0.0..=1.0).contains(&weight) {
        return Err((StatusCode::BAD_REQUEST, Json(error_body("VALIDATION_ERROR", "weight must be within 0..=1"))));
    }
    let id = state.node_seq.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
    let orchestrator = &state.orchestrator;
    orchestrator.upsert_node(kce_core::types::EcmaNode {
        id,
        state: kce_core::types::NodeState::Stem,
        usage_count: 1,
        entropy: 1.0,
        connections: 0,
        maturity: 0.0,
    });
    if let Some(target) = req.connect_to {
        if !orchestrator.connect(id, target, weight) {
            orchestrator.remove_node(id);
            return Err((StatusCode::NOT_FOUND, Json(error_body("NOT_FOUND", "connect_to node not found"))));
        }
    }
    let _ = state.events.send(crate::events::EngineEvent::NodeCreated { id });
    Ok((StatusCode::CREATED, Json(CreateNodeResponse { id })))
}

/// Remove a node from the graph.
async fn delete_node_handler(

/// Deposit pheromone on an edge.
async fn deposit_handler(
    State(state): State<Arc<AppState>>,
    Json(req): Json<DepositRequest>,
) -> Result<(StatusCode, Json<DepositResponse>), (StatusCode, Json<crate::routes::ErrorBody>)> {
    if !req.amount.is_finite() || req.amount <= 0.0 {
        return Err((StatusCode::BAD_REQUEST, Json(error_body("VALIDATION_ERROR", "amount must be finite and > 0"))));

        .peek_fractional(req.from, req.to, 0)
        .unwrap_or(0.0);
    let _ = state.events.send(crate::events::EngineEvent::PheromoneDeposited {
        from: req.from,
        to: req.to,
        level,
